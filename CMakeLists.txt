cmake_minimum_required(VERSION 3.16)
project(L1pp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(L1pp L1pp/main.cpp L1pp/integration.cpp L1pp/posix.cpp L1pp/std_thread.cpp)
if(WIN32)
  if(NOT MSVC)
    message(FATAL_ERROR "On Windows use MSVC with supplied pthreadVC2 library")
  endif()
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use x64")
  endif()
  target_include_directories(L1pp PRIVATE pthread/Pre-built.2/include)
  target_link_libraries(L1pp PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/pthread/Pre-built.2/lib/x64/pthreadVC2.lib")
  target_compile_options(L1pp PRIVATE /utf-8 /W4)
  add_custom_command(TARGET L1pp POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_SOURCE_DIR}/pthread/Pre-built.2/dll/x64/pthreadVC2.dll" "$<TARGET_FILE_DIR:L1pp>")
else()
  find_package(Threads REQUIRED)
  target_link_libraries(L1pp PRIVATE Threads::Threads)
  target_compile_options(L1pp PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME correctness COMMAND L1pp --self-test)
